getBoundingClientRect()

بترجعلك إحداثيات الـ element بالنسبة للـ viewport. الـ hook بيستخدم rect.left وrect.right وrect.bottom عشان يحسب فين بالظبط هيحط الـ dropdown تحت الـ button من غير ما يخرج برا الشاشة.

// الثوابت const dropdownWidth = 240; // w-60 = 15rem // الـ initial position let left = rect.left + window.scrollX; const top = rect.bottom + window.scrollY; // Fix 1 — overflow من اليمين if (left + dropdownWidth > window.innerWidth) { left = rect.right + window.scrollX - dropdownWidth; // Fix 2 — fallback لو لسه بيخرج if (left > 0) { left = window.innerWidth - dropdownWidth - 16; } // Fix 3 — clamp — يمنع الخروج من الشمال if (left < 0) { left = 16; } }
